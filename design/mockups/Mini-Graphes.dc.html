règle 1 · pas de glyphes JetBrains Mono (Google Fonts) ne contient pas les blocs Unicode. La police de secours les dessine plus bas et décalés : on les remplace par des SVG.
règle 2 · taille en em 0.6em de large par barre (une colonne de caractère), 0.75em de haut. Le graphe suit la taille du texte et reste sur la ligne de base.
règle 3 · currentColor Le SVG hérite de la couleur du texte (fill="currentColor"). Les cases vides des jauges sont en #45475A.
règle 4 · décoratif aria-hidden="true" sur chaque graphe. La valeur réelle (genre, %, bpm) est toujours écrite en texte à côté.

// heights : tableau d'entiers 1 → 8
export function MiniGraph({ heights, className }) {
  const n = heights.length;
  const d = heights
    .map((h, i) => `M${8 * i} ${8 - h}h7v${h}h-7z`)
    .join('');
  return (
    <svg aria-hidden="true" viewBox={`0 0 ${8 * n - 1} 8`}
      preserveAspectRatio="none" fill="currentColor"
      className={className}
      style={ { width: `${0.6 * n}em`, height: '0.75em' } }>
      <path d={d} />
    </svg>
  );
}

// jauge : value 0 → 1, cells = 10 (ou 8 pour le volume)
export function Meter({ value, cells = 10 }) {
  const on = Math.round(value * cells);
  const cell = (i) => `M${8 * i} 0h7v8h-7z`;
  const range = (a, b) =>
    Array.from({ length: b - a }, (_, k) => cell(a + k)).join('');
  return (
    <svg aria-hidden="true" viewBox={`0 0 ${8 * cells - 1} 8`}
      preserveAspectRatio="none"
      style={ { width: `${0.6 * cells}em`, height: '0.75em' } }>
      <path fill="currentColor" d={range(0, on)} />
      <path fill="#45475A" d={range(on, cells)} />
    </svg>
  );
}
